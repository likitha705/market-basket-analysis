def normalize_name(name: str) -> str:
    """Lowercase, strip whitespace, collapse punctuation."""
    name = str(name).lower().strip()
    name = re.sub(r"[-_/\\]+", " ", name)
    name = re.sub(r"\s+", " ", name)
    return name

def load_and_map_columns(df: pd.DataFrame, col_map: dict) -> pd.DataFrame:
    """Rename columns according to user mapping."""
    return df.rename(columns={v: k for k, v in col_map.items() if v})
def validate_dataframe(df: pd.DataFrame) -> tuple[bool, str]:
    if df.empty:
        return False, "The uploaded file is empty."
    missing = REQUIRED_COLS - set(df.columns)
    if missing:
        return False, f"Missing required columns: {', '.join(missing)}"
    if df["transaction_id"].isna().all():
        return False, "Column 'transaction_id' contains no valid values."
    if df["product_name"].isna().all():
        return False, "Column 'product_name' contains no valid values."
    return True, "OK"

def preprocess(df: pd.DataFrame) -> tuple[pd.DataFrame, dict]:
    """Clean and normalise the transaction data. Returns cleaned df + stats."""
    stats = {}
    stats["raw_rows"] = len(df)
    stats["raw_transactions"] = df["transaction_id"].nunique()
    stats["raw_products"] = df["product_name"].nunique()
    # Drop rows with null transaction_id or product_name
    df = df.dropna(subset=["transaction_id", "product_name"])
    # Normalise product names
    df = df.copy()
    df["product_name_raw"] = df["product_name"]
    df["product_name"] = df["product_name"].apply(normalize_name)
    # Drop duplicate (transaction_id, product_name) pairs
    before_dedup = len(df)
    df = df.drop_duplicates(subset=["transaction_id", "product_name"])
    stats["duplicate_rows_removed"] = before_dedup - len(df)
    # Drop transactions with no products
    txn_counts = df.groupby("transaction_id")["product_name"].count()
    valid_txns = txn_counts[txn_counts > 0].index
    df = df[df["transaction_id"].isin(valid_txns)]
    stats["clean_rows"] = len(df)
    stats["clean_transactions"] = df["transaction_id"].nunique()
    stats["clean_products"] = df["product_name"].nunique()
    return df, stats

def build_similarity_graph(
    products: list[str], threshold: float = 80.0
) -> nx.Graph:
    """
    Build a product-similarity graph using RapidFuzz token_sort_ratio.
    Nodes = product names; edges = similarity ≥ threshold.
    """
    G = nx.Graph()
    G.add_nodes_from(products)
    if not RAPIDFUZZ_OK:
        st.warning("RapidFuzz not installed — similarity graph unavailable.")
        return G
    for i, p1 in enumerate(products):
        for p2 in products[i + 1:]:
            score = fuzz.token_sort_ratio(p1, p2)
            if score >= threshold:
                G.add_edge(p1, p2, weight=round(score / 100, 3), score=score)
    return G
def find_duplicate_groups(G: nx.Graph) -> list[list[str]]:
    """Return connected components with more than one node."""
    return [
        sorted(list(c))
        for c in nx.connected_components(G)
        if len(c) > 1
    ]
def pick_canonical(group: list[str]) -> str:
    """Choose the shortest, most 'clean' name as the canonical representative."""
    return min(group, key=lambda x: (len(x), x))
def build_dedup_mapping(groups: list[list[str]]) -> dict[str, str]:
    """Map every duplicate product name to its canonical representative."""
    mapping = {}
    for group in groups:
        canonical = pick_canonical(group)
        for name in group:
            mapping[name] = canonical
    return mapping
def apply_dedup(df: pd.DataFrame, mapping: dict[str, str]) -> pd.DataFrame:
    """Replace product names with canonical names from the dedup mapping."""
    df = df.copy()
    df["product_name"] = df["product_name"].apply(lambda n: mapping.get(n, n))
    # Drop newly created (transaction_id, product_name) duplicates
    df = df.drop_duplicates(subset=["transaction_id", "product_name"])
    return df

def build_transaction_matrix(df: pd.DataFrame) -> pd.DataFrame:
    """One-hot encoded transaction-item matrix."""
    basket = df.groupby(["transaction_id", "product_name"])["product_name"].count().unstack().fillna(0)
    basket = basket.applymap(lambda x: True if x > 0 else False)
    return basket
def mine_frequent_itemsets(
    basket: pd.DataFrame,
    algorithm: str = "fpgrowth",
    min_support: float = 0.05,
) -> pd.DataFrame:
    if not MLXTEND_OK:
        st.error("mlxtend is not installed. Cannot mine frequent itemsets.")
        return pd.DataFrame()
    try:
        if algorithm == "apriori":
            fi = apriori(basket, min_support=min_support, use_colnames=True)
        else:
            fi = fpgrowth(basket, min_support=min_support, use_colnames=True)
        if fi.empty:
            return fi
        fi["itemset_size"] = fi["itemsets"].apply(len)
        fi["items"] = fi["itemsets"].apply(lambda x: ", ".join(sorted(x)))
        fi = fi.sort_values("support", ascending=False).reset_index(drop=True)
        return fi
    except Exception as e:
        st.error(f"Error during frequent itemset mining: {e}")
        return pd.DataFrame()
def generate_association_rules_df(
    fi: pd.DataFrame,
    min_confidence: float = 0.3,
    min_lift: float = 1.0,
) -> pd.DataFrame:
    if not MLXTEND_OK or fi.empty:
        return pd.DataFrame()
    try:
        rules = association_rules(fi, metric="confidence", min_threshold=min_confidence)
        rules = rules[rules["lift"] >= min_lift]
        rules["antecedents_str"] = rules["antecedents"].apply(lambda x: ", ".join(sorted(x)))
        rules["consequents_str"] = rules["consequents"].apply(lambda x: ", ".join(sorted(x)))
        rules = rules.sort_values("lift", ascending=False).reset_index(drop=True)
        return rules
    except Exception as e:
        st.error(f"Error generating association rules: {e}")
        return pd.DataFrame()

#  CO-PURCHASE GRAPH
# ============================================================================
def build_copurchase_graph(df: pd.DataFrame) -> nx.Graph:
    """
    Build a co-purchase graph.
    Nodes = products (sized by frequency); edges = # shared transactions.
    """
    G = nx.Graph()
    product_freq = df.groupby("product_name")["transaction_id"].nunique().to_dict()
    for product, freq in product_freq.items():
        G.add_node(product, frequency=freq)
    txn_groups = df.groupby("transaction_id")["product_name"].apply(list)
    for products_in_txn in txn_groups:
        for p1, p2 in combinations(products_in_txn, 2):
            if G.has_edge(p1, p2):
                G[p1][p2]["weight"] += 1
            else:
                G.add_edge(p1, p2, weight=1)
    return G

def pyvis_similarity_graph(G: nx.Graph, height: int = 500) -> str:
    """Render similarity graph with PyVis and return HTML string."""
    if not PYVIS_OK:
        return ""
    net = Network(height=f"{height}px", width="100%", bgcolor="#1a1a2e", font_color="white")
    net.barnes_hut(gravity=-5000, central_gravity=0.3, spring_length=120)
    # Colour by component
    components_list = list(nx.connected_components(G))
    color_palette = [
        "#a78bfa", "#60a5fa", "#34d399", "#f59e0b",
        "#f87171", "#fb923c", "#e879f9", "#38bdf8",
    ]
    node_color = {}
    for idx, comp in enumerate(components_list):
        color = color_palette[idx % len(color_palette)] if len(comp) > 1 else "#64748b"
        for n in comp:
            node_color[n] = color
    for node in G.nodes():
        degree = G.degree(node)
        net.add_node(
            node,
            label=node,
            color=node_color.get(node, "#64748b"),
            size=15 + degree * 4,
            title=f"Product: {node}<br>Connections: {degree}",
        )
    for u, v, data in G.edges(data=True):
        score = data.get("score", 0)
        net.add_edge(
            u, v,
            value=data.get("weight", 1),
            title=f"Similarity: {score:.0f}%",
            color=f"rgba(167,139,250,{min(1.0, score/100)})",
        )
    net.set_options("""
    {
      "interaction": {"hover": true, "tooltipDelay": 100},
      "physics": {"stabilization": {"iterations": 100}}
    }
    """)
    with tempfile.NamedTemporaryFile(delete=False, suffix=".html", mode="w", encoding="utf-8") as f:
        net.save_graph(f.name)
        return open(f.name, "r", encoding="utf-8").read()
def pyvis_copurchase_graph(
    G: nx.Graph,
    min_weight: int = 1,
    category_filter: list[str] | None = None,
    height: int = 600,
) -> str:
    """Render co-purchase graph with PyVis and return HTML string."""
    if not PYVIS_OK:
        return ""
    net = Network(height=f"{height}px", width="100%", bgcolor="#1a1a2e", font_color="white")
    net.barnes_hut(gravity=-6000, central_gravity=0.3, spring_length=150)
    # Filter edges
    filtered_edges = [
        (u, v, d) for u, v, d in G.edges(data=True) if d.get("weight", 0) >= min_weight
    ]
    if not filtered_edges:
        return ""
    active_nodes = set()
    for u, v, _ in filtered_edges:
        active_nodes.add(u)
        active_nodes.add(v)
    max_freq = max((G.nodes[n].get("frequency", 1) for n in active_nodes), default=1)
    max_weight = max((d.get("weight", 1) for _, _, d in filtered_edges), default=1)
    for node in active_nodes:
        freq = G.nodes[node].get("frequency", 1)
        size = 12 + (freq / max_freq) * 30
        net.add_node(
            node,
            label=node,
            size=size,
            color="#6366f1",
            title=f"<b>{node}</b><br>Transaction frequency: {freq}",
            font={"size": 12},
        )
    for u, v, data in filtered_edges:
        w = data.get("weight", 1)
        thickness = 1 + (w / max_weight) * 6
        net.add_edge(
            u, v,
            value=thickness,
            title=f"Co-purchased in {w} transaction(s)",
            color=f"rgba(96,165,250,{min(0.9, 0.2 + w / max_weight * 0.7)})",
        )
    net.set_options("""
    {
      "interaction": {"hover": true, "tooltipDelay": 100},
      "physics": {"stabilization": {"iterations": 150}}
    }
    """)
    with tempfile.NamedTemporaryFile(delete=False, suffix=".html", mode="w", encoding="utf-8") as f:
        net.save_graph(f.name)
        return open(f.name, "r", encoding="utf-8").read()
def plotly_similarity_graph_fallback(G: nx.Graph) -> go.Figure:
    """Plotly fallback when PyVis unavailable."""
    pos = nx.spring_layout(G, seed=42)
    edge_x, edge_y = [], []
    for u, v in G.edges():
        x0, y0 = pos[u]; x1, y1 = pos[v]
        edge_x += [x0, x1, None]; edge_y += [y0, y1, None]
    edge_trace = go.Scatter(
        x=edge_x, y=edge_y, mode="lines",
        line=dict(width=1, color="rgba(167,139,250,0.5)"),
        hoverinfo="none",
    )
    node_x = [pos[n][0] for n in G.nodes()]
    node_y = [pos[n][1] for n in G.nodes()]
    node_text = list(G.nodes())
    node_trace = go.Scatter(
        x=node_x, y=node_y, mode="markers+text",
        marker=dict(size=14, color="#a78bfa", line=dict(width=1, color="white")),
        text=node_text, textposition="top center",
        hoverinfo="text",
    )
    fig = go.Figure(
        data=[edge_trace, node_trace],
        layout=go.Layout(
            paper_bgcolor="rgba(0,0,0,0)",
            plot_bgcolor="rgba(26,26,46,1)",
            font=dict(color="white"),
            showlegend=False,
            margin=dict(l=20, r=20, t=20, b=20),
            xaxis=dict(showgrid=False, zeroline=False, showticklabels=False),
            yaxis=dict(showgrid=False, zeroline=False, showticklabels=False),
        ),
    )
    return fig
def plotly_copurchase_fallback(G: nx.Graph, min_weight: int = 1) -> go.Figure:
    """Plotly fallback co-purchase graph."""
    H = nx.Graph()
    for u, v, d in G.edges(data=True):
        if d.get("weight", 0) >= min_weight:
            H.add_edge(u, v, weight=d["weight"])
    for n, d in G.nodes(data=True):
        if n in H:
            H.nodes[n]["frequency"] = d.get("frequency", 1)
    if H.number_of_edges() == 0:
        return go.Figure()
    pos = nx.spring_layout(H, seed=42)
    edge_x, edge_y = [], []
    for u, v in H.edges():
        x0, y0 = pos[u]; x1, y1 = pos[v]
        edge_x += [x0, x1, None]; edge_y += [y0, y1, None]
    edge_trace = go.Scatter(
        x=edge_x, y=edge_y, mode="lines",
        line=dict(width=1, color="rgba(96,165,250,0.4)"),
        hoverinfo="none",
    )
    freqs = [H.nodes[n].get("frequency", 1) for n in H.nodes()]
    max_f = max(freqs) if freqs else 1
    node_trace = go.Scatter(
        x=[pos[n][0] for n in H.nodes()],
        y=[pos[n][1] for n in H.nodes()],
        mode="markers+text",
        marker=dict(
            size=[10 + 20 * (f / max_f) for f in freqs],
            color=freqs,
            colorscale="Viridis",
            showscale=True,
            colorbar=dict(title="Frequency", titlefont=dict(color="white"), tickfont=dict(color="white")),
            line=dict(width=1, color="white"),
        ),
        text=list(H.nodes()),
        textposition="top center",
        hoverinfo="text",
    )
    fig = go.Figure(
        data=[edge_trace, node_trace],
        layout=go.Layout(
            paper_bgcolor="rgba(0,0,0,0)",
            plot_bgcolor="rgba(26,26,46,1)",
            font=dict(color="white"),
            showlegend=False,
            margin=dict(l=20, r=20, t=20, b=20),
            xaxis=dict(showgrid=False, zeroline=False, showticklabels=False),
            yaxis=dict(showgrid=False, zeroline=False, showticklabels=False),
        ),
    )
    return fig

def generate_insights(rules: pd.DataFrame, fi: pd.DataFrame, df: pd.DataFrame) -> list[str]:
    insights = []
    if rules.empty:
        insights.append("No association rules found. Try lowering minimum support or confidence.")
        return insights
    # Top rule by lift
    top = rules.iloc[0]
    insights.append(
        f"**🏆 Strongest Association:** Customers who buy **{top['antecedents_str']}** "
        f"also tend to buy **{top['consequents_str']}** "
        f"(Lift: {top['lift']:.2f}, Confidence: {top['confidence']:.0%})."
    )
    # Cross-sell
    high_conf = rules[rules["confidence"] >= 0.6]
    if not high_conf.empty:
        r = high_conf.iloc[0]
        insights.append(
            f"**🔁 Cross-sell Opportunity:** Place **{r['consequents_str']}** "
            f"near **{r['antecedents_str']}** — "
            f"{r['confidence']:.0%} of buyers purchase both."
        )
    # Bundle suggestions
    size2 = fi[fi["itemset_size"] == 2].head(3)
    if not size2.empty:
        pairs = size2["items"].tolist()
        insights.append(
            f"**📦 Bundle Promotions:** Consider bundling — "
            + "; ".join([f"*{p}*" for p in pairs]) + "."
        )
    # Most frequent product
    top_product = df.groupby("product_name")["transaction_id"].nunique().idxmax()
    insights.append(
        f"**⭐ Best-Seller:** **{top_product.title()}** appears in the most transactions. "
        "Ensure it is always well-stocked and prominently displayed."
    )
    # High-lift rules for placement
    placement = rules[rules["lift"] > 2].head(2)
    for _, r in placement.iterrows():
        insights.append(
            f"**🗺️ Shelf Placement:** Products **{r['antecedents_str']}** and "
            f"**{r['consequents_str']}** have lift {r['lift']:.2f} — "
            "placing them in adjacent aisles may increase basket size."
        )

                | **Confidence** | P(B \| A) = P(A ∪ B) / P(A) | Of all transactions with A, how many also have B |
                | **Lift** | Confidence / P(B) | >1 means A and B appear together more than by chance |
